SELECT ACTOR_NAME, DATEDIFF('YEAR', ACTOR_BIRTH_DATE, ACTOR_DEAD_DATE) AS EDAD
FROM ACTORS
WHERE DATEDIFF('YEAR', ACTOR_BIRTH_DATE, ACTOR_DEAD_DATE) <= 50

--11 devuelve el nombre de todos los directores menores o iguales de 40 años que esten vivos 
SELECT DIRECTOR_NAME 
FROM DIRECTORS d 
WHERE DATEDIFF('YEAR', d.DIRECTOR_BIRTH_DATE, TODAY()) <= 40 AND DIRECTOR_DEAD_DATE IS NULL

--12 INDICA LA EDAD MEDIA DE LOS DIRECTORES VIVOS 
SELECT AVG(DATEDIFF ('YEAR', DIRECTOR_BIRTH_DATE, TODAY())) AS EDAD_PROMEDIO 
FROM DIRECTORS 
WHERE DIRECTOR_DEAD_DATE IS NULL

--13 INDICA LA EDAD MEDIA DE LOS ACTORES QUE HAN FALLECIDO 
SELECT AVG(DATEDIFF ('YEAR',ACTOR_BIRTH_DATE, ACTOR_DEAD_DATE)) AS EDAD_MEDIA 
FROM ACTORS 
WHERE ACTOR_DEAD_DATE IS NOT NULL

WITH NATIONALITY_COUNTS AS
 
(SELECT s.STUDIO_ID, s.studio_name, n.NATIONALITY_NAME, COUNT(m.NATIONALITY_ID ) AS NUM_MOVIES FROM MOVIES m
INNER JOIN STUDIOS s ON m.STUDIO_ID =s.STUDIO_ID
 
INNER JOIN NATIONALITIES n  ON m.NATIONALITY_ID =n.NATIONALITY_ID
 
GROUP BY s.STUDIO_ID, n.nationality_name),
 
MAX_NATIONALITIES AS (SELECT studio_ID, MAX(NUM_MOVIES) AS MAX_MOVIES FROM NATIONALITY_COUNTS
 
GROUP BY studio_id )
 
SELECT nc.STUDIO_NAME, GROUP_CONCAT(NATIONALITY_NAME SEPARATOR ', ') AS FAVORITE_NATIONALITY FROM NATIONALITY_COUNTS nc
 
INNER JOIN MAX_NATIONALITIES mn ON nc.STUDIO_ID = mn.STUDIO_ID AND nc.NUM_MOVIES = mn.MAX_MOVIES
 
GROUP BY nc.STUDIO_NAME;
